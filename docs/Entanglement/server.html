server.py is responsible for sending messages between users. The messages are simple, they're just strings contianing a single word. To be a touch banal, if i tell the server 'red', all the server does is retell this to whomever it may concern.

        import asyncio
        import websockets
        import random
        import json
        import uuid


        connected_clients = {}
        Entangled = True
        Opposite = True
        collapsed = False


        collapsed_clients = []
        collapseTo = 0
        groups = []

        pair_states = {}
        collapsed_values = {}

        def get_groups():
            clients = list(connected_clients.keys())[1:]
            return [clients[i:i+2] for i in range(0, len(clients), 2)]

        def get_partner(websocket):
            for group in get_groups():
                if websocket in group:
                    if len(group) == 2:
                        return group[1] if group[0] is websocket else group[0]
                    return None
            return None

        def get_pair_key(ws_a, ws_b):
            id_a = connected_clients[ws_a]["id"]
            id_b = connected_clients[ws_b]["id"]
            return tuple(sorted([id_a, id_b]))

        async def ws_handler(websocket):
            global Entangled, Opposite, collapsed_clients, collapseTo, collapsed, superUserId
            client_id = str(uuid.uuid4())[:8]
            connected_clients[websocket] = {
                "id": client_id
            }

            await websocket.send(json.dumps({
                "type": "state_init",
                "entangled": Entangled,
                "opposite": Opposite,
                "connected_clients": len(connected_clients) - 1,
                "id": connected_clients[websocket]["id"]
            }))

            try:
                async for message in websocket:
                    print(f"Received: {message}")
                    data = json.loads(message)

                    if data.get("type") == "coin_clicked":
                        sender_ws = websocket
                        sender_id = connected_clients[sender_ws]["id"]

                        partner_ws = get_partner(sender_ws)
                        if partner_ws is None:
                            await sender_ws.send(json.dumps({
                                "type": "coin_clicked",
                                "inGroup": False
                            }))
                            continue

                        partner_id = connected_clients[partner_ws]["id"]
                        pair_key = get_pair_key(sender_ws, partner_ws)

                        first_measurement = False

                        if Entangled:
                            if pair_key not in pair_states:
                                base = random.choice([0, 1])
                                pair_states[pair_key] = {"collapseTo": base}
                                value = base
                                first_measurement = True
                            else:
                                base = pair_states[pair_key]["collapseTo"]
                                value = 1 - base if Opposite else base
                        else:
                            value = random.choice([0, 1])

                        collapsed_values[sender_id] = value

                        await sender_ws.send(json.dumps({
                            "type": "coin_clicked",
                            "inGroup": True,
                            "partner": partner_id,
                            "value": value,
                            "firstMeasurement": first_measurement
                        }))
                        
                    if data.get("type") == "entanglement-toggle":
                            Entangled = not Entangled
                            for client, data in connected_clients.items():
                                try:
                                    await client.send(json.dumps({
                                        "type": "entangled"
                                    }))
                                except Exception as e:
                                    print(f"Error sending to client {data['id']}: {e}")

                    if data.get("type") == "opposite-toggle":
                            Opposite = not Opposite
                            for client, data in connected_clients.items():
                                try:
                                    await client.send(json.dumps({
                                        "type": "opposite"
                                    }))
                                except Exception as e:
                                    print(f"Error sending to client {data['id']}: {e}")
                    
                    if data.get("type") == "reset-all":
                        pair_states.clear()
                        collapsed_values.clear()
                        for client, info in connected_clients.items():
                            try:
                                await client.send(json.dumps({"type": "reset"}))
                            except Exception as e:
                                print(f"Error sending to client {info['id']}: {e}")

                    if data.get("type") == "super":
                        Entangled = True
                        Opposite = True
                        pair_states.clear()
                        collapsed_values.clear()

                    # if data.get("type") == "super":
                    #     Entangled = True
                    #     Opposite = True
                    #     collapsed = False
                    #     collapseTo = 0
                    #     collapsed_clients = []


                    if data.get("type") == "request-id-list":
                        sender_id = connected_clients[websocket]["id"]
                        client_list = []
                        for client_ws, client_info in connected_clients.items():
                            if client_info["id"] == sender_id:
                                continue
                            value = collapsed_values.get(client_info["id"], "Superposition")
                            client_list.append({"id": client_info["id"], "value": value})
                        await websocket.send(json.dumps({
                            "type": "recievedIDList",
                            "clients": client_list
                        }))

                    if data.get("type") == "disconnect-all":
                        websockets_to_close = list(connected_clients.keys())


                        for websocket in websockets_to_close:

                            await websocket.send(json.dumps({"action": "force_reload"}))
                            await asyncio.sleep(0.1)

                            try:
                                await websocket.close(code=4001, reason="Server shutting down or force disconnect")
                                print(f"Disconnected client: {connected_clients[websocket]['id']}")
                                del connected_clients[websocket]
                            except Exception as e:
                                print(f"Error disconnecting a client: {e}")
                            


            except websockets.ConnectionClosed:
                print("Client disconnected")
            finally:
                client_info = connected_clients.pop(websocket, None)
                if client_info:
                    leaving_id = client_info["id"]
                    collapsed_values.pop(leaving_id, None)
                    to_delete = []
                    for key in pair_states.keys():
                        if leaving_id in key:
                            to_delete.append(key)
                    for key in to_delete:
                        del pair_states[key]

        async def main():
            async with websockets.serve(ws_handler, "0.0.0.0", 8765):
                print("WebSocket server started on ws://0.0.0.0:8765")
                await asyncio.Future()

        if __name__ == "__main__":
            asyncio.run(main())ubuntu@nyfiken:~$
