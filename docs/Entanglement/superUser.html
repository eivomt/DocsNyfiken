Super user

Super user has control over the settings for the visualization for each connected user. Possible actions for the super user are toggling opposite and entangled. Setting all users back to superposition and disconnecting all users.

super user can at all times see how many clients are connected and their status (collapsed?).

Users are paired in two's so that the measurement of one user is entangled only to whomever was paired up with them initially. The pairs are made chronologically, a user is paired with the next one to connect or the previous depending on the number of already connected clients (odd / even). If a user is not connected to another, collapse is disallowed. The number preceding user id's in the list of connected clients represents what pair the user belongs to.

which returns an array of hashmaps containing the id of all connected clients (except the superuser) along with their values (0 for blue, 1 for red).         connectButton.addEventListener('click', () => {
            socket = new WebSocket('wss://nyfiken.cs.oslomet.no/ws/')

            socket.addEventListener('open', () => {
                console.log('Connected to server!')
                appendMessage("Connected to server.")
            })

            socket.addEventListener('message', (event) => {
                let data = JSON.parse(event.data)

                if (data.action === 'force_reload') {
                    console.log('Server requested page reload');
                    window.location.reload(true);
                }

                switch (data.type) {
                    case 'state_init':
                        entangled = data.entangled
                        opposite = data.opposite

                        document.getElementById("connectButton").style.display = "none"

                        console.log('entangled = ' + entangled)
                        console.log('opposite = ' + opposite)

                        console.log('user id = ' + data.id)

                        console.log('# of connected clients: ' + data.connected_clients)


                        console.log(data.connected_clients)
                        if (data.connected_clients == 0) {
                            superUser = true
                            becomeSuper()
                        }
                        break

                    case 'coin_clicked':
                        console.log('data.value = ' + data.value)
                        if (!superUser) {
                            console.log('Measured first')
                            console.log('Recieved: ' + data.value + ' from server')
                            if (data.value === 1) {
                                console.log('Displaying red')
                                cutToRed()
                            } else if (data.value === 0) {
                                console.log('Displaying blue')
                                cutToBlue()
                            }
                        }
                    break
                    
                    case 'entangled':
                        entangled = !entangled
                        console.log('State changed, Entangled = ' + entangled)
                    break

                    case 'opposite':
                        opposite = !opposite
                        console.log('State changed, opposite = ' + opposite)
                    break

                    case 'reset':
                        cutToWhite()
                        coin.addEventListener("click", sendCoinClick)

                    case 'recievedIDList':
                        recievedID(data.clients)
                }
            });

            socket.addEventListener('close', () => {
                console.log('Disconnected from server')
                appendMessage("Disconnected from server.")
                document.getElementById("connectButton").style.display = "block"
            });

            socket.addEventListener('error', (error) => {
                console.error('WebSocket error:', error)
            });
        });
