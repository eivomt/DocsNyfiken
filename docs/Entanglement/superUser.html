        connectButton.addEventListener('click', () => {
            socket = new WebSocket('wss://nyfiken.cs.oslomet.no/ws/')

            socket.addEventListener('open', () => {
                console.log('Connected to server!')
                appendMessage("Connected to server.")
            })

            socket.addEventListener('message', (event) => {
                let data = JSON.parse(event.data)

                if (data.action === 'force_reload') {
                    console.log('Server requested page reload');
                    window.location.reload(true);
                }

                switch (data.type) {
                    case 'state_init':
                        entangled = data.entangled
                        opposite = data.opposite

                        document.getElementById("connectButton").style.display = "none"

                        console.log('entangled = ' + entangled)
                        console.log('opposite = ' + opposite)

                        console.log('user id = ' + data.id)

                        console.log('# of connected clients: ' + data.connected_clients)


                        console.log(data.connected_clients)
                        if (data.connected_clients == 0) {
                            superUser = true
                            becomeSuper()
                        }
                        break

                    case 'coin_clicked':
                        console.log('data.value = ' + data.value)
                        if (!superUser) {
                            console.log('Measured first')
                            console.log('Recieved: ' + data.value + ' from server')
                            if (data.value === 1) {
                                console.log('Displaying red')
                                cutToRed()
                            } else if (data.value === 0) {
                                console.log('Displaying blue')
                                cutToBlue()
                            }
                        }
                    break
                    
                    case 'entangled':
                        entangled = !entangled
                        console.log('State changed, Entangled = ' + entangled)
                    break

                    case 'opposite':
                        opposite = !opposite
                        console.log('State changed, opposite = ' + opposite)
                    break

                    case 'reset':
                        cutToWhite()
                        coin.addEventListener("click", sendCoinClick)

                    case 'recievedIDList':
                        recievedID(data.clients)
                }
            });

            socket.addEventListener('close', () => {
                console.log('Disconnected from server')
                appendMessage("Disconnected from server.")
                document.getElementById("connectButton").style.display = "block"
            });

            socket.addEventListener('error', (error) => {
                console.error('WebSocket error:', error)
            });
        });
