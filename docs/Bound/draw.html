psiTotal is the array that stores the visible wave function curve. More precisely, it stores the value that is drawn at each horizontal position on the canvas. In the code, the real and imaginary parts of the wave function are first calculated, and then combined into a probability-like value: 

            psiTotal.push((amplitudeScalar*temporaryReal**2) + amplitudeScalar*temporaryImaginary**2)
So psiTotal is not the complex wave function itself, but the drawable height of the probability distribution.

collapsed is a boolean variable that decides whether the system is currently shown as a superposition or as one measured energy state. When collapsed is false, the program adds together all energy states. When it is true, the program only draws one selected energy state.

collapseTo is used to make sure the measurement happens only once when the system collapses. Without it, the program would choose a new random energy level on every frame. Since draw() runs continuously, that would make the state jump constantly. When collapse starts, collapseTo is set to true. The program then chooses one energy level, stores it in energy, updates the measurement bars, and sets: 

            collapseTo = false
After that, the same measured state continues to be drawn.

resizeMeasurements() updates the visual measurement bars. Each bar has a stored count of how many times that energy level has been measured. The function reads those counts, normalizes them, and then changes the height of each bar: 

            measurements[i].style.height = (measurementsArray[i] * sumInput).toString() + "%"
This means that more frequently measured energy levels get taller bars.

measurement.dataset refers to custom data stored directly on the HTML element. In this code, each measurement bar has a value called data-measurements. 

            <div class="measurement-outer-container">
                <div class="measurement-container">
                    <div class="slider">
                        <div class="measurement" data-measurements="0"></div>
                    </div>
                </div>
                <math-field class="bar-label" read-only virtual-keyboard-mode="off" menu-items="[]">E_{0}</math-field>
            </div>
JavaScript reads it like this: 
            measurements[i].dataset.measurements
and updates it like this: 
            measurements[energy].dataset.measurements =
            ((parseInt(measurements[energy].dataset.measurements)) + 1).toString()
So the HTML element itself keeps track of how many times that energy level has been selected.

The block 

            if(collapsed) {
                ...
            } else {
                ...
            }
separates the two main behaviours of the simulation. If the system is collapsed, the program draws only the measured energy state. If the system is not collapsed, the program draws the full superposition by summing over all energy states: 
            for(let j=0; j<EigValues.length; j++) {
                temporaryReal += ... * amplitudeArray[j]
                temporaryImaginary += ... * amplitudeArray[j]
            }
In short: if(collapsed) shows the result after measurement, while else shows the unmeasured superposition. 




        function draw() {
            let sliderInputs = []

            for(let i=0; i <EigValues.length; i++) {
                sliderInputs.push(document.getElementById("E" + (i+1).toString()))
            }

            recalculateProbabilities(sliderInputs)


            if(paused==false) {
                let bgColor = color('#F1F1F1')
                background(bgColor)
                translate(-512 + windowWidth/2, windowHeight/2 + 4*windowHeight/32)


                drawingContext.setLineDash([1,4])
                strokeWeight(2)
                stroke(255,255,255)
                for(let i = 0; i<EigValues.length; i++){
                    line(0,-30*EigValues[i], 512*2,-30*EigValues[i])
                }

                

                strokeWeight(2)
            
                drawingContext.setLineDash([1])
                stroke(32,115,232)
                // stroke('#333')
                beginShape()
                for(let i = 0; i<potential.length; i++) {
                    vertex(i*2, -potential[i])
                }
                endShape()

                drawingContext.setLineDash([1])
                stroke(252,65,32)
                
                noFill()
                strokeWeight(3)
                beginShape()
                for(let i = 0; i< psiTotal.length; i++) {
                    vertex(i*2,-psiTotal[i])
                }
                endShape()
            

                

                t += .04
                psiTotal = []
                totalReal = []
                totalImaginary = []
            
                
                
                // reverse loop 
                for(let i=0; i < EigVectorsReal.length; i++) {
                    temporaryReal = 0
                    temporaryImaginary = 0

                    
            
                    if(collapsed) { 
                        if(collapseTo) {

                            if(burst) {
                                for (let i = 0; i < 4; i ++) {
                                    let iterator = 0
                                    energy = Math.random()
                                    while (energy > amplitudePositions[iterator]) {
                                        iterator++
                                    }
                                    energy = iterator

                                    let measurements = document.querySelectorAll(".measurement")

                                    measurements[energy].dataset.measurements = ((parseInt(measurements[energy].dataset.measurements)) + 1).toString()
                                    resizeMeasurements(measurements)
                                }
                                burst = false
                            }

                            let iterator = 0
                            energy = Math.random()
                            while (energy > amplitudePositions[iterator]) {
                                iterator++
                            }
                            energy = iterator

                            const energyLevelID = `lE${energy}`
                            gsap.to(`#${energyLevelID}`, {x: -20, duration: .25})
                            gsap.to(`#${energyLevelID}`, {scale: 2.5, duration: .25, delay: .25})
                            gsap.to(`#${energyLevelID}`, {scale: 1, duration: .35, delay: 1.5})
                            gsap.to(`#${energyLevelID}`, {x: 0, duration: .25, delay: 1.75})

                            let measurements = document.querySelectorAll(".measurement")

                            measurements[energy].dataset.measurements = ((parseInt(measurements[energy].dataset.measurements)) + 1).toString()

                            resizeMeasurements(measurements)
                            collapseTo = false
                        }
                        // pause loop
                        temporaryReal += (Math.cos(EigValues[energy]*t) * EigVectorsReal[i][energy] - Math.sin(EigValues[energy]*t)*EigVectorsImaginary[i][energy])
                        temporaryImaginary += (Math.cos( EigValues[energy]*t) * EigVectorsImaginary[i][energy] + Math.sin(EigValues[energy]*t)*EigVectorsReal[i][energy])
                    } else { 
                        for(let j=0; j<EigValues.length; j++) {
                            // * initial coefficients?
                            temporaryReal += (Math.cos(EigValues[j]*t) * EigVectorsReal[i][j] - Math.sin(EigValues[j]*t)*EigVectorsImaginary[i][j]) * amplitudeArray[j]
                            temporaryImaginary += (Math.cos( EigValues[j]*t) * EigVectorsImaginary[i][j] + Math.sin(EigValues[j]*t)*EigVectorsReal[i][j]) * amplitudeArray[j]
                        }
                    }
                    totalReal.push(temporaryReal)
                    totalImaginary.push(temporaryImaginary)
            
                    if(collapsed) {
                        psiTotal.push((((amplitudeScalar/EigValues.length)*temporaryReal**2) + (amplitudeScalar/EigValues.length)*temporaryImaginary**2) + EigValues[energy] * 30)
                    } else {
                        psiTotal.push((amplitudeScalar*temporaryReal**2) + amplitudeScalar*temporaryImaginary**2)
                    }
                }
            }
